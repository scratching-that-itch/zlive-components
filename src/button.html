@zig { /// shadcn/ui's six variants, same names. pub const Variant = enum { @"default", outline, secondary, ghost, destructive, link, pub fn tag(self: Variant) []const u8 { return @tagName(self); } }; /// shadcn/ui's eight sizes, same names (the icon-* sizes are square). pub const Size = enum { @"default", xs, sm, lg, icon, @"icon-xs", @"icon-sm", @"icon-lg", pub fn tag(self: Size) []const u8 { return @tagName(self); } }; } @props { label: []const u8, /// Visual variant. `default` is spelled `@"default"` here because /// `default` is a Zig keyword; the rendered class/data-variant is /// "default". variant: Variant = .@"default", size: Size = .@"default", disabled: bool = false, /// Renders type="submit". (shadcn passes `type` straight through; a /// bool is the Zig-friendly spelling of the case that matters.) submit: bool = false, /// Extra classes, appended after the variant and size classes — the /// same merge shadcn's `className` prop does. Use it for layout hooks /// ("detail-close", an anchor class), not to restyle a variant. class: ?[]const u8 = null, /// Element id. Required when the button is a Floating UI anchor /// (data-zc-anchor resolves it with getElementById) or when a hook /// targets it. id: ?[]const u8 = null, /// event name sent on click; null renders no data-lv-click attribute on_click: ?[]const u8 = null, /// data-lv-value: the client wraps it as {id, value}, and it wins over /// data-lv-payload when both are present. value: ?[]const u8 = null, /// data-lv-payload: raw JSON the client parses and sends verbatim as /// the event payload. payload: ?[]const u8 = null, /// aria-expanded. Null omits the attribute (most buttons); true/false /// render it, which is what a disclosure trigger (dropdown, popover) /// needs to stay accessible while the server owns the open state. aria_expanded: ?bool = null, /// aria-label, for icon-only buttons whose visible label is decorative. aria_label: ?[]const u8 = null, }
