@zig { /// The icons this library draws, inlined so a page fetches nothing at /// runtime and a changed icon is an ordinary segment patch. /// /// Shapes are lucide's (lucide-icons, ISC licence), 24x24, stroked with /// `currentColor` at the width lucide uses, so an icon inherits the text /// colour of whatever it sits in. `tag()` is the lucide name. pub const Name = enum { panel_left, chevrons_up_down, plus, square_terminal, bot, book_open, settings_2, chevron_right, frame, pie_chart, map, more_horizontal, folder, forward, trash_2, sparkles, badge_check, credit_card, bell, log_out, gallery_vertical_end, audio_waveform, command, pub fn tag(self: Name) []const u8 { return @tagName(self); } }; } @props { name: Name, /// Extra classes, appended — the same merge `Button.class` does (sizes /// come from the context: a sidebar menu button's icon is 1rem). class: ?[]const u8 = null, }
